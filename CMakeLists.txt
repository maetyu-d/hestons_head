cmake_minimum_required(VERSION 3.20)
project(Aftertaste VERSION 3.28.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(AFTERTASTE_BUILD_APP "Build the native desktop application" ON)
option(AFTERTASTE_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(aftertaste_core src/Model.cpp src/Simulation.cpp src/Scoring.cpp src/Forecast.cpp src/Persistence.cpp src/Factory.cpp src/FactoryFloor.cpp src/Culinary.cpp src/FloorTelemetry.cpp src/Power.cpp src/PowerDistrict.cpp src/NuclearLab.cpp src/WaterWorks.cpp src/ReagentWorks.cpp src/Demo.cpp src/Control.cpp src/ControlDemo.cpp)
target_include_directories(aftertaste_core PUBLIC src vendor)
target_compile_options(aftertaste_core PRIVATE -Wall -Wextra -Wpedantic)
if(AFTERTASTE_SANITIZE)
  target_compile_options(aftertaste_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(aftertaste_core PUBLIC -fsanitize=address,undefined)
endif()
if(AFTERTASTE_BUILD_APP)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  add_subdirectory(vendor/raylib EXCLUDE_FROM_ALL)
  if(APPLE)
    add_executable(Aftertaste MACOSX_BUNDLE src/Main.cpp src/Interface.cpp src/FactoryInterface.cpp src/WorkspacePanels.cpp src/FloorRendering.cpp src/MachineArt.cpp src/PowerArt.cpp src/RouteArt.cpp src/PowerInterface.cpp src/GlassArt.cpp src/ControlArt.cpp src/ControlUI.cpp)
    set_target_properties(Aftertaste PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "Aftertaste"
      MACOSX_BUNDLE_GUI_IDENTIFIER "studio.aftertaste.reactor"
      MACOSX_BUNDLE_BUNDLE_VERSION "3.28.0" MACOSX_BUNDLE_SHORT_VERSION_STRING "3.28.0"
      MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in")
  else()
    add_executable(Aftertaste src/Main.cpp src/Interface.cpp src/FactoryInterface.cpp src/WorkspacePanels.cpp src/FloorRendering.cpp src/MachineArt.cpp src/PowerArt.cpp src/RouteArt.cpp src/PowerInterface.cpp src/GlassArt.cpp src/ControlArt.cpp src/ControlUI.cpp)
  endif()
  target_link_libraries(Aftertaste PRIVATE aftertaste_core raylib)
  target_compile_options(Aftertaste PRIVATE -Wall -Wextra -Wpedantic)
  if(APPLE)
    add_custom_command(TARGET Aftertaste POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_CONTENT_DIR:Aftertaste>/Resources"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/README.txt" "$<TARGET_BUNDLE_CONTENT_DIR:Aftertaste>/Resources/README.txt"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/VALIDATION.txt" "$<TARGET_BUNDLE_CONTENT_DIR:Aftertaste>/Resources/VALIDATION.txt"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/vendor/json-LICENSE.txt" "$<TARGET_BUNDLE_CONTENT_DIR:Aftertaste>/Resources/json-LICENSE.txt"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/vendor/raylib/LICENSE" "$<TARGET_BUNDLE_CONTENT_DIR:Aftertaste>/Resources/raylib-LICENSE.txt")
  endif()
  install(TARGETS Aftertaste BUNDLE DESTINATION . RUNTIME DESTINATION bin)
endif()
